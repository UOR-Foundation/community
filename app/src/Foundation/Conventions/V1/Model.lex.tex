definition{
  name: measuredOf,
  parameters: [measure : Measure, repository : Repository],
  result: Nat,
  body: (match measure
    | Measure.versions => (((project repository tagsVersion) + (project repository tagsInvalid)) + (project repository manifestsVersioned))
    | Measure.commits => (project repository commitsEvaluated)
    | Measure.changelogStructure => (1 + (project repository changelogChecks))
    | Measure.changelogFile => 1
    | Measure.releaseNotes => (project repository tagsVersion))
}

definition{
  name: failingOf,
  parameters: [measure : Measure, repository : Repository],
  result: Nat,
  body: (match measure
    | Measure.versions => ((project repository tagsInvalid) + (project repository manifestsInvalid))
    | Measure.commits => (project repository nonconforming)
    | Measure.changelogStructure => (if (project repository changelog) then (project repository changelogFailed) else 1)
    | Measure.changelogFile => (if (project repository changelog) then 0 else 1)
    | Measure.releaseNotes => 0)
}

definition{
  name: statusFits,
  parameters: [status : Status, observable : Bool, measured : Nat, failing : Nat],
  result: Bool,
  body: (if observable then ((((isMet status) && ((0 <? measured) && (failing =? 0))) || ((isUnmet status) && (0 <? failing))) || ((isVacuous status) && (measured =? 0))) else (((isVacuous status) && (measured =? 0)) || ((isUnobserved status) && (0 <? measured))))
}
