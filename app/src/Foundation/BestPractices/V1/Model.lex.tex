  parameters: [criterion : Criterion, outcome : Outcome],
  result: Bool,

definition{
  name: agreesHead,
  parameters: [cursor : (List Criterion), position : Nat, outcome : Outcome],
  result: Bool,
  body: (match cursor
    | List.nil => false
    | List.cons criterion rest => (((project outcome repository) =? position) && (outcomeAgrees criterion outcome)))
}

definition{
  name: blockAligned,
  parameters: [
    whole : (List Criterion),
    cursor : (List Criterion),
    position : Nat,
    width : Nat,
    values : (List Outcome)
  ],
  result: Bool,
  body: (match values
    | List.nil => (position =? width)
    | List.cons outcome rest => ((agreesHead cursor position outcome) && (blockAligned whole (advance whole cursor) (nextPosition cursor position) width rest))),
  recursive_argument: values
}
