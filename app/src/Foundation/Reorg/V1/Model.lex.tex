definition{

}

definition{
  name: ruleGives,
  parameters: [position : Nat, disposition : Disposition],
  result: Bool,
  body: (match disposition
    | Disposition.keep => ((position =? 2) || (position =? 0))
    | Disposition.archive => ((position =? 1) || (position =? 4))
    | Disposition.fold => (position =? 3)
    | Disposition.review => ((position =? 5) || ((position =? 6) || ((position =? 7) || ((position =? 8) || (position =? 9))))))
}
