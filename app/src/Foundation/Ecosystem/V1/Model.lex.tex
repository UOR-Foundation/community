structure{
  name: Release,
  type_parameters: [],
  parameters: [],
  fields: [

definition{
  name: repositoriesValid,
  parameters: [position : Nat, values : (List Repository)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons repository rest => (((project repository index) =? position) && (repositoriesValid (position + 1) rest))),
  recursive_argument: values
}

definition{
  name: repositoriesTotal,
  parameters: [values : (List Repository)],
  result: Nat,
  body: (match values
    | List.nil => 0
    | List.cons repository rest => (1 + (repositoriesTotal rest))),
  recursive_argument: values
}

definition{
  name: openClaimsValid,
  parameters: [position : Nat, repositoriesCount : Nat, values : (List OpenClaim)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons openClaim rest => (((project openClaim index) =? position) && (((project openClaim repository) <? repositoriesCount) && (openClaimsValid (position + 1) repositoriesCount rest)))),
  recursive_argument: values
}

definition{
  name: openClaimsTotal,
  parameters: [values : (List OpenClaim)],
  result: Nat,
  body: (match values | List.nil => 0 | List.cons openClaim rest => (1 + (openClaimsTotal rest))),
  recursive_argument: values
}

definition{
  name: tocApplicationsValid,
  parameters: [position : Nat, values : (List TocApplication)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons tocApplication rest => (((project tocApplication index) =? position) && (tocApplicationsValid (position + 1) rest))),
  recursive_argument: values
}

definition{
  name: tocApplicationsTotal,
  parameters: [values : (List TocApplication)],
  result: Nat,
  body: (match values
    | List.nil => 0
    | List.cons tocApplication rest => (1 + (tocApplicationsTotal rest))),
  recursive_argument: values
}

definition{
  name: tierRecorded,
  parameters: [value : Tier],
  result: Bool,
  body: (match value
    | Tier.tier1Core => true
    | Tier.tier2Impl => true
    | Tier.tier3Presentation => true
    | Tier.tier4Experimental => true
    | Tier.unrecorded => false)
}

definition{
  name: statusRecorded,
  parameters: [value : Status],
  result: Bool,
  body: (match value
    | Status.active => true
    | Status.maintenance => true
    | Status.deprecated => true
    | Status.archived => true
    | Status.unrecorded => false)
}

definition{
  name: isConformant,
  parameters: [value : StandardsOutcome],
  result: Bool,
  body: (match value
    | StandardsOutcome.untagged => false
    | StandardsOutcome.unversioned => false
    | StandardsOutcome.conformant => true
    | StandardsOutcome.undecided => false)
}

definition{
  name: agree,
  parameters: [left : Bool, right : Bool],
  result: Bool,
  body: (if left then right else (not right))
}

definition{
  name: tagsRead,
  parameters: [repository : Repository],
  result: Bool,
  body: ((agree (tierRecorded (project repository tier)) ((project repository tierTags) =? 1)) && (agree (statusRecorded (project repository status)) ((project repository statusTags) =? 1)))
}

definition{
  name: conformantTotal,
  parameters: [values : (List Repository)],
  result: Nat,
  body: (match values
    | List.nil => 0
    | List.cons repository rest => ((if (isConformant (project repository outcome)) then 1 else 0) + (conformantTotal rest))),
  recursive_argument: values
}

definition{
  name: claimed,
  parameters: [position : Nat, claims : (List OpenClaim)],
  result: Bool,
  body: (match claims
    | List.nil => false
    | List.cons claim rest => (((project claim repository) =? position) || (claimed position rest))),
  recursive_argument: claims
}

definition{
  name: repositoriesAccounted,
  parameters: [claims : (List OpenClaim), values : (List Repository)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons repository rest => (((tagsRead repository) && ((isConformant (project repository outcome)) || (claimed (project repository index) claims))) && (repositoriesAccounted claims rest))),
  recursive_argument: values
}

definition{
  name: registerValid,
  parameters: [
    release : Release,
    repositories : (List Repository),
    openClaims : (List OpenClaim),
    tocApplications : (List TocApplication)
  ],
  result: Bool,
  body: (((repositoriesTotal repositories) =? (project release repositoriesCount)) && (((openClaimsTotal openClaims) =? (project release openClaimsCount)) && (((tocApplicationsTotal tocApplications) =? (project release tocApplicationsCount)) && ((repositoriesValid 0 repositories) && ((openClaimsValid 0 (repositoriesTotal repositories) openClaims) && ((tocApplicationsValid 0 tocApplications) && (((conformantTotal repositories) =? (project release conformantCount)) && (repositoriesAccounted openClaims repositories))))))))
}
