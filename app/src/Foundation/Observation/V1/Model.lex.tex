definition{
  name: admissionDecided,
  parameters: [observation : Observation],
  result: Bool,
  body: (if (metricStatusIsUnreleased (project observation metricStatus)) then (admissionIsNotApplicable (project observation admission)) else (if (collectionStatusIsDeny (project observation collectionStatus)) then (admissionIsDeny (project observation admission)) else (if (collectionStatusIsOutside (project observation collectionStatus)) then (admissionIsDeny (project observation admission)) else (if (attributionStatusIsDeny (project observation attributionStatus)) then (admissionIsDeny (project observation admission)) else (if (oracleStatusIsExcludes (project observation oracleStatus)) then (admissionIsDeny (project observation admission)) else (if (sourceStatusIsUnpinned (project observation sourceStatus)) then (admissionIsDeny (project observation admission)) else (if (replayStatusIsDiffers (project observation replayStatus)) then (admissionIsDeny (project observation admission)) else (if (oracleStatusIsUnlisted (project observation oracleStatus)) then (admissionIsIndeterminate (project observation admission)) else (if (replayStatusIsAbsent (project observation replayStatus)) then (admissionIsIndeterminate (project observation admission)) else (if (collectionStatusIsIndeterminate (project observation collectionStatus)) then (admissionIsIndeterminate (project observation admission)) else (if (attributionStatusIsIndeterminate (project observation attributionStatus)) then (admissionIsIndeterminate (project observation admission)) else (admissionIsPermit (project observation admission)))))))))))))
}

definition{
  name: admittedTotal,
  parameters: [values : (List Observation)],
  result: Nat,
  body: (match values
    | List.nil => 0
    | List.cons observation rest => ((if (admissionIsPermit (project observation admission)) then 1 else 0) + (admittedTotal rest))),
  recursive_argument: values
}

definition{
  name: skipTo,
  parameters: [position : Nat, observations : (List Observation)],
  result: (List Observation),
  body: (match observations
    | List.nil => (nil Observation)
    | List.cons observation rest => (if ((project observation index) <? position) then (skipTo position rest) else (cons observation rest))),
  recursive_argument: observations
}

}
