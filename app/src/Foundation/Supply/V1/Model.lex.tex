definition{
  name: componentsOf,
  parameters: [position : Nat, values : (List ComponentCount)],
  result: Nat,
  body: (match values
    | List.nil => 0
    | List.cons row rest => (if ((project row repository) =? position) then ((project row count) + (componentsOf position rest)) else (componentsOf position rest))),
  recursive_argument: values
}

definition{
  name: countsReconcile,
  parameters: [
    allCounts : (List ComponentCount),
    allLicenses : (List LicenseCount),
    values : (List Repository)

  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons repository rest => ((((componentsOf (project repository index) allCounts) =? (project repository components)) && ((licensesOf (project repository index) allLicenses) =? (project repository licenseEntries))) && (countsReconcile allCounts allLicenses rest))),

definition{
  name: sbomsAccounted,
  parameters: [claims : (List RepositoryClaim), values : (List Repository)],
  result: Bool,
  body: (match values
    | List.nil => true

  recursive_argument: values

definition{
  name: registryAccepted,
  parameters: [status : RegistryStatus, isPrivate : Bool],
  result: Bool,
  body: (match status

    | RegistryStatus.absent => isPrivate
    | RegistryStatus.elsewhere => false
    | RegistryStatus.unread => false)
}

definition{
  name: packagesAccounted,
  parameters: [claims : (List PackageClaim), values : (List Package)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons item rest => (((registryAccepted (project item registry) (project item declaredPrivate)) || (packageClaimed (project item index) claims)) && (packagesAccounted claims rest))),
  recursive_argument: values
}
