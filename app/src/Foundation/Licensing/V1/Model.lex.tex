definition{
  name: agree,
  parameters: [left : Bool, right : Bool],
  result: Bool,
  body: (if left then right else (not right))
}

definition{
  name: noFindings,
  parameters: [repository : Repository],
  result: Bool,
  body: ((((project repository missingLicenses) =? 0) && (((project repository badLicenses) =? 0) && (((project repository deprecatedLicenses) =? 0) && ((project repository unusedLicenses) =? 0)))) && ((((project repository licensesWithoutExtension) =? 0) && (((project repository readErrors) =? 0) && ((project repository invalidExpressions) =? 0))) && (((project repository filesWithCopyright) =? (project repository filesTotal)) && ((project repository filesWithLicense) =? (project repository filesTotal)))))
}

definition{
  name: countsReconcile,
  parameters: [repository : Repository],
  result: Bool,
  body: ((agree (project repository compliant) (noFindings repository)) && ((((project repository filesWithCopyright) <? ((project repository filesTotal) + 1)) && ((project repository filesWithLicense) <? ((project repository filesTotal) + 1))) && ((project repository identifiedFiles) <? ((project repository licenseFiles) + 1))))
}

  name: compliantTotal,
  parameters: [values : (List Repository)],
  result: Nat,
  body: (match values
    | List.nil => 0

  recursive_argument: values
}

definition{
  name: repositoriesClaimed,

(project repository licenseFiles)) && (((project repository identifiedFiles) =? (project repository licenseFiles)) || (claimedLicenseTextUnidentified (project repository index) claims))))) && (repositoriesClaimed claims rest))),
  recursive_argument: values
}

  recursive_argument: values
}

(expectedOf allRepositories (project declaration repository)))) else true) && (conformingNamesExpected allIds allRepositories rest))),
  recursive_argument: values
}

(project release repositoriesCount)) && (((declarationsTotal declarations) =? (project release declarationsCount)) && (((declaredIdsTotal declaredIds) =? (project release declaredIdsCount)) && (((openClaimsTotal openClaims) =? (project release openClaimsCount)) && (((policyClaimsTotal policyClaims) =?
